                    POST
                    curl -X POST -d 'hello' http://127.0.0.1:8080/upload/test.txt
                    
                

                

                    console
                    fetch('/upload/test.txt', {method:'POST', body:'hello'}).then(r =>
                        r.text()).then(console.log)
                    
                

            

            

                

                    PUT
                    curl -X PUT -d 'updated' http://127.0.0.1:8080/upload/test.txt
                    
                

                

                    console
                    fetch('/upload/test.txt', {method:'PUT', body:'updated'}).then(r =>
                        r.text()).then(console.log)

                    curl http://127.0.0.1:8080/upload/test.txt
                    
                

                

                    console
                    fetch('/upload/test.txt').then(r => r.text()).then(console.log)
